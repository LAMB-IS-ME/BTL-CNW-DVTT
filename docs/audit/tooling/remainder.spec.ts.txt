function save(name:string,data:unknown){writeFileSync(`${out}/${name}.json`,JSON.stringify(data,null,2)+'\n');}
async function mutate(c:BrowserContext,p:string,data:any={},method='POST'){
 const token=(await(await c.request.get(base+'/auth/csrf')).json()).data.token;
 const r=await c.request.fetch(base+p,{method,data,headers:{Origin:origin,'X-CSRF-Token':token}});
 expect(r.ok(),`${method} ${p}: ${await r.text()}`).toBeTruthy();
 return r.status()===204?null:(await r.json()).data;
}
async function login(c:BrowserContext,email:string){return mutate(c,'/auth/login',{email,password:process.env.DEMO_PASSWORD});}
